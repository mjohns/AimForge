set(NAME "aim")

set(PROTO_FILES
    "common.proto"
    "bundle.proto"
    "playlist.proto"
    "theme.proto"
    "scenario.proto"
    "stats.proto"
    "settings.proto"
    "crosshair.proto"
)
set(PROTO_DIR "${CMAKE_CURRENT_SOURCE_DIR}/proto/schema/")
list(TRANSFORM PROTO_FILES PREPEND "${PROTO_DIR}" OUTPUT_VARIABLE FULL_PROTO_FILES)
message(STATUS "Full proto file paths: ${FULL_PROTO_FILES}")

file(GLOB_RECURSE SRCS *.cc)
file(GLOB_RECURSE HDRS *.h)

add_library(${NAME} ${SRCS} ${HRDS} ${FULL_PROTO_FILES})

set(PROTO_BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
protobuf_generate(
    TARGET ${NAME}
    IMPORT_DIRS "${CMAKE_CURRENT_LIST_DIR}/proto/schema"
    PROTOC_OUT_DIR "${PROTO_BINARY_DIR}/aim/proto")

target_link_libraries(
    ${NAME}
    PUBLIC
    SDL3::SDL3
    SDL3_mixer::SDL3_mixer
    SDL_image
    absl::strings
    absl::flags
    glm::glm
    imgui
    implot
    sqlite3
    protobuf::libprotobuf
    spdlog)

target_include_directories(${NAME} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..  "$<BUILD_INTERFACE:${PROTO_BINARY_DIR}>")

set_target_properties(${NAME} PROPERTIES COMPILE_WARNING_AS_ERROR ON)

if(MSVC)
    # Disables warnings from protobuf lib
    target_compile_options(${NAME} PRIVATE /wd4091)
    # /we4062 treats missing enum cases as errors but allows default
    target_compile_options(${NAME} PRIVATE /we4062)
endif()
